cmake_minimum_required(VERSION 3.30)
project(BlenderCoreDependencyTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenEXR CONFIG REQUIRED)
find_package(PNG REQUIRED)
find_package(JPEG REQUIRED)
find_package(TIFF REQUIRED)
find_package(OpenColorIO CONFIG REQUIRED)
find_package(OpenImageIO CONFIG REQUIRED)
find_package(Freetype REQUIRED)
find_library(BROTLI_COMMON brotlicommon REQUIRED)
find_library(BROTLI_DEC brotlidec REQUIRED)
add_executable(core-dependencies-probe ../../tests/core_dependencies_probe.cc)
target_link_libraries(core-dependencies-probe PRIVATE OpenImageIO::OpenImageIO
  OpenColorIO::OpenColorIO Freetype::Freetype ${BROTLI_DEC} ${BROTLI_COMMON})
